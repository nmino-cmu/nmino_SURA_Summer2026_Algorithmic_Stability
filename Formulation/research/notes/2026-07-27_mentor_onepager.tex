\noindent\textit{Scope ceiling (Wenbin):} UQ~(1)(2); selected object \(\hat\theta(\mathcal{D})\) only;
goal~(iii) $=$ data-randomize selection (\textbf{not} recalibrate \(\mathcal{C}\)).

\paragraph{Setup.}
Fix classical level \(\alpha\) and sets \(\mathcal{C}^{(\alpha)}(X;\mathcal{D},\theta)\)
(writeup~(1) arity: third slot $=\theta$; \(\alpha$ only as level / superscript --- not in the third slot).
Let \(\tilde\theta(\mathcal{D})\) come from constrained opt~(2) after a randomized selection design
(RNM / soft-argmin). If \(\tilde\theta\) is \((\eta,0,\nu)\)-stable (ZJ), then
\[
\mathrm{Infl}
:=
P\bigl(Y\notin\mathcal{C}^{(\alpha)}(X;\mathcal{D},\tilde\theta)\bigr)-\alpha
\;\le\;
(e^\eta-1)\alpha+\nu.
\]
Sending \(\eta,\nu\to0\) restores~(1) at the \emph{same} \(\alpha\) and the \emph{same} map \(\mathcal{C}\).

\paragraph{Part I (design).}
Concentrate \((\hat f,\hat g)\) at rate \(\varepsilon(n,\nu)\) (DKW).
Choose Lap scale \(b=2\varepsilon/\eta\) or soft \(\beta=\eta/(2\varepsilon)\)
\(\Rightarrow\) \((\eta,0,\nu)\)-stable \(\tilde\theta\), \textbf{restricted to}
\(\Theta_{\mathrm{val}}\) (score-threshold: \(\theta\ge F^{-1}(1-\alpha)\); tradeoff: conservative \(\Theta_0^\varepsilon\)).
\(\eta\to0\) costs selection utility (not free).

\paragraph{Instances of~(2).}
\textbf{(2$_{\mathrm{nest}}$)}: \(\hat f=1-\widehat F_n\), box \(\hat g\) --- hard argmin collapses to \(\theta=1\); live object is randomized.
\textbf{(2$_{\mathrm{trade}}$)}: \(\hat f(\theta)=\theta\), \(\hat g=\widehat{\mathrm{misc}}-\alpha_0\) --- hard \(\hat\theta=\) empirical quantile (non-degenerate).
Constraint $\alpha_0$ is not the classical level $\alpha$ of~(1); numerics set $\alpha_0=\alpha$.

\paragraph{Part II.}
Set-valued ZJ transfer under~(OV)+(ST) (writeup~(1) on $\mathrm{supp}(A_0)$;
for score-threshold sets, only $\theta\ge F^{-1}(1-\alpha)$): keep classical level $\alpha$ fixed;
bound Infl as above. Recalibrating to level $\alpha e^{-\eta}$ answers a different question (rejected by W5).
Notation: $C^{(\alpha)}$ $=$ writeup $\mathcal{C}$ at level $\alpha$; third slot remains $\theta$.
