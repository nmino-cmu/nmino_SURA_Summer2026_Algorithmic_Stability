\begin{abstract}
We study \(\mathrm{sign}(x)\in\{-1,0,1\}\) under \(\lvert x'-x\rvert\le\varepsilon\).
Positive/negative signs are preserved when \(\lvert x\rvert>\varepsilon\); the origin is
stable only for \(\varepsilon=0\). Lean \texttt{LEAN\_FULL} on Mathlib \(\mathbb{R}\)
(\texttt{REAL\_MATHLIB}).
\end{abstract}
\section{Operator}
\begin{definition}
\(\mathrm{sign}(x)=1\) if \(x>0\), \(-1\) if \(x<0\), and \(0\) if \(x=0\).
\end{definition}
\section{Theorems}
\begin{theorem}[Preservation]
If \(\lvert x'-x\rvert\le\varepsilon\): (1) \(x>\varepsilon\Rightarrow\mathrm{sign}(x')=1\);
(2) \(x<-\varepsilon\Rightarrow\mathrm{sign}(x')=-1\);
(3) \(\varepsilon=0\land x=0\Rightarrow\mathrm{sign}(x')=0\).
\end{theorem}
\begin{theorem}[Sharpness]
If \(0<x\le\varepsilon\) then some admissible \(x'\) has \(\mathrm{sign}(x')\ne 1\);
symmetrically on the negative side; if \(x=0\) and \(\varepsilon>0\) then some
\(x'\) has \(\mathrm{sign}(x')\ne 0\).
\end{theorem}
\section{Proofs}
Preservation follows from \(\lvert x'-x\rvert\le\varepsilon\) and \(\omega\)-arithmetic.
Sharpness uses \(x'=x\pm\varepsilon\) or \(x'=\varepsilon\) at the origin.
