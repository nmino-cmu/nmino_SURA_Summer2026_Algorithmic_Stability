\section{Problem}
{\raggedright\sloppy
The sign operator returns $\mathrm{sign}(x)=+1$ if $x>0$, $-1$ if $x<0$, and $0$ if $x=0$.
\par}

\section{Stability notion}
Perturbation: $|x'-x|\le\varepsilon$. Preservation uses \emph{strict} buffers $\varepsilon<x$ and $x<-\varepsilon$. Zero output is preserved only when $\varepsilon=0$ and $x=0$. Sharpness covers $(0,\varepsilon]$, $[-\varepsilon,0)$, and $x=0$ with $\varepsilon>0$.

\section{Definitions}
\begin{definition}[sign]
$\mathrm{sign}(x)=1$ if $0<x$; $-1$ if $x<0$; else $0$.
\end{definition}

\section{Theorem}
\begin{theorem}[Sign preservation]\label{thm:inv}
If $|x'-x|\le\varepsilon$, then: (i) $\varepsilon<x\Rightarrow\mathrm{sign}(x')=1$; (ii) $x<-\varepsilon\Rightarrow\mathrm{sign}(x')=-1$; (iii) $\varepsilon=0\land x=0\Rightarrow\mathrm{sign}(x')=0$.
\end{theorem}
\begin{theorem}[Sharpness]\label{thm:sharp}
If $0<x\le\varepsilon$ then some $|x'-x|\le\varepsilon$ has $\mathrm{sign}(x')\neq 1$; symmetrically for $-\varepsilon\le x<0$; if $x=0$ and $\varepsilon>0$ then some $|x'-x|\le\varepsilon$ has $\mathrm{sign}(x')\neq 0$.
\end{theorem}

\section{Intuition}
A strict buffer larger than $\varepsilon$ keeps the perturbed point away from $0$. On the closed band the push $x\mapsto x-\varepsilon$ can leave the positive ray.

\section{Examples}
\begin{example}
$x=2$, $\varepsilon=0.5$: $\varepsilon<x$, so the sign stays $+1$. $x=0.4$, $\varepsilon=0.5$: $x'=x-\varepsilon=-0.1$ has sign $-1$.
\end{example}

\section{Proof}
From $|x'-x|\le\varepsilon$, $x-\varepsilon\le x'\le x+\varepsilon$. If $\varepsilon<x$ then $x'>0$, so $\mathrm{sign}(x')=1$. If $x<-\varepsilon$ then $x'<0$, so $\mathrm{sign}(x')=-1$. If $\varepsilon=0$ and $x=0$ then $x'=0$. This is Theorem~\ref{thm:inv}.

Sharpness: if $0<x\le\varepsilon$, take $x'=x-\varepsilon\le 0$. If $-\varepsilon\le x<0$, take $x'=x+\varepsilon\ge 0$. If $x=0$ and $\varepsilon>0$, take $x'=\varepsilon$. This is Theorem~\ref{thm:sharp}.
