\begin{abstract}
Scalar threshold $A_T(x)=\mathbf{1}\{x\ge T\}$ is invariant outside the $2\varepsilon$ unstable band when $|x'-x|\le\varepsilon$. Lean \texttt{LEAN\_FULL}.
\end{abstract}

\section{Problem}
{\raggedright\sloppy
The operator returns the binary indicator $A_T(x)=1$ iff $x\ge T$ (equality passes).
\par}

\section{Stability notion}
Perturbation: $|x'-x|\le\varepsilon$. Pass buffer $x\ge T+\varepsilon\Rightarrow A_T(x')=1$; fail buffer $x<T-\varepsilon\Rightarrow A_T(x')=0$.

\section{Definitions}
\begin{definition}
$A_T(x)\Leftrightarrow x\ge T$.\end{definition}
\noindent\textbf{Assumptions.} $\varepsilon\ge 0$ and $|x'-x|\le\varepsilon$.

\section{Theorem}
\begin{theorem}\label{thm:inv}
If $|x'-x|\le\varepsilon$, then $x\ge T+\varepsilon\Rightarrow x'\ge T$ and $x<T-\varepsilon\Rightarrow x'<T$.
\end{theorem}

\section{Intuition}
Moving $x$ by at most $\varepsilon$ cannot cross $T$ from outside the $\varepsilon$-neighborhood.

\section{Examples}
\begin{example}
$T=3$, $x=5$, $\varepsilon=0.2$: $x\ge T+\varepsilon$, so every $x'$ in the ball still passes.
\end{example}

\section{Proof}
From $|x'-x|\le\varepsilon$ one has $x'-\varepsilon\le x\le x'+\varepsilon$. If $x\ge T+\varepsilon$ then $x'\ge x-\varepsilon\ge T$. If $x<T-\varepsilon$ then $x'\le x+\varepsilon<T$ (Theorem~\ref{thm:inv}).
