\section{Problem}
{\raggedright\sloppy
The noisy threshold $\widetilde A_T(x)=\mathbf{1}\{x+\xi\ge T\}$ with $|\xi|\le\eta$ a.s. The selected object is the binary pass/fail bit.
\par}

\section{Stability notion}
Noise is query-side and bounded. Outside $[T-\eta,T+\eta)$ the output is a.s.\ constant; inside the band it need not be.

\section{Definitions}
Pass convention: $x\ge T$ passes. Pathwise: apply deterministic threshold preservation with $\varepsilon=\eta$.

\section{Theorem}
\begin{theorem}\label{thm:inv}
If $|\xi|\le\eta$ a.s., then $x\ge T+\eta\Rightarrow\widetilde A_T=1$ a.s.\ and $x<T-\eta\Rightarrow\widetilde A_T=0$ a.s. On the open band with $\eta>0$, two-point noise shows the bit need not be a.s.\ constant.
\end{theorem}

\section{Intuition}
Bounded noise cannot cross a buffer of width $\eta$ on either side of $T$.

\section{Examples}
\begin{example}
$T=3$, $x=5$, $\eta=0.2$: $x\ge T+\eta$, so the bit stays $1$ a.s.
\end{example}

\section{Proof}
On almost every outcome, $|\xi(\omega)|\le\eta$. Apply deterministic threshold preservation to $(x,T,\varepsilon,x')=(x,T,\eta,x+\xi(\omega))$: if $x\ge T+\eta$ then $x+\xi\ge T$; if $x<T-\eta$ then $x+\xi<T$. For the band claim, the two-point law on $\{\pm\eta\}$ realizes both sides of the cut (Theorem~\ref{thm:inv}). This is a pathwise surrogate of a.s.\ bounded noise, not a differential-privacy accounting proof.
