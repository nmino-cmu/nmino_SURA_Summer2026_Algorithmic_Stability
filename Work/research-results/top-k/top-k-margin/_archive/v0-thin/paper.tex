\begin{abstract}
Top-$k$ selection is certified via pairwise ranking preservation when all gaps
exceed $2\varepsilon$. Lean: \texttt{LEAN\_FULL}.
\end{abstract}
\section{Theorems}
\begin{theorem}[Preservation]
All pairwise gaps $>2\varepsilon$ imply $s_i<s_j$ iff $(s+\delta)_i<(s+\delta)_j$ for $\|\delta\|_\infty\le\varepsilon$, hence the top-$k$ index set is invariant.
\end{theorem}
\begin{theorem}[Sharpness]
A pairwise gap $\le 2\varepsilon$ admits a collision adversary.
\end{theorem}
