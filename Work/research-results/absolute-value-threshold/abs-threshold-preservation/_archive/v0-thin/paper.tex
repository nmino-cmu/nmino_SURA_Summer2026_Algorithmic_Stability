\begin{abstract}
We study \(A(x)=\mathbf{1}\{\lvert x\rvert\ge T\}\) for \(T\ge 0\) under
\(\lvert x'-x\rvert\le\varepsilon\). Pass/fail buffers use
\(\lvert x\rvert\ge T+\varepsilon\) and \(\lvert x\rvert+\varepsilon<T\).
Lean \texttt{LEAN\_FULL} on Mathlib \(\mathbb{R}\) (\texttt{REAL\_MATHLIB}).
\end{abstract}
\section{Theorems}
\begin{theorem}[Preservation]
If \(\lvert x'-x\rvert\le\varepsilon\): (1) \(\lvert x\rvert\ge T+\varepsilon\Rightarrow A(x')=1\);
(2) \(\lvert x\rvert+\varepsilon<T\Rightarrow A(x')=0\).
\end{theorem}
\begin{theorem}[Sharpness]
If \(\varepsilon\le x\) and \(T\le x<T+\varepsilon\), then \(x'=x-\varepsilon\) fails \(A\).
\end{theorem}
