\begin{abstract}
The kth order statistic operator selects a unique strict $k$-th order-statistic index. When every pairwise score gap exceeds $2\varepsilon$, that index is invariant under $\|\delta\|_\infty\le\varepsilon$. Lean: \texttt{LEAN\_FULL} (reduction of \texttt{OrderStat.KthMargin}).

\section{Problem}
{\raggedright\sloppy
The \leanpath{kth-order-statistic} operator on scores $s\in\mathbb{R}^n$ ($n\ge 2$) returns the unique index $i$ that is the strict $k$-th smallest coordinate of $s$ (operator-specific choice of $k$). The selected object is this index under a unique strict-order presentation.
\par}

\section{Stability notion}
Perturbations are additive $\delta$ with $\|\delta\|_\infty\le\varepsilon$. Stability: if $i$ is the unique strict $k$-th smallest and $\mathrm{AllGapsExceed}(s,2\varepsilon)$, then $i$ remains the unique strict $k$-th smallest of $s+\delta$. Sharpness: a rival within distance $2\varepsilon$ admits an admissible tying adversary.

\section{Definitions}
\begin{definition}[Strict $k$-th smallest]
Index $i$ is strict $k$-th smallest if $\#\{j:s_j<s_i\}=k$ and $s_j=s_i\Rightarrow j=i$.
\end{definition}
\begin{definition}[All-gaps]
$|s_p-s_q|>g$ for all $p\neq q$.
\end{definition}
\noindent\textbf{Assumptions.} $n\ge 2$, $\varepsilon\ge 0$, unique strict $k$-th index, and (for invariance) all pairwise gaps $>2\varepsilon$.

\section{Theorem}
\begin{theorem}[Invariance]\label{thm:inv}
If $i$ is strict $k$-th smallest and $\mathrm{AllGapsExceed}(s,2\varepsilon)$, then for every $\|\delta\|_\infty\le\varepsilon$, $i$ remains strict $k$-th smallest for $s+\delta$.
\end{theorem}
\begin{theorem}[Sharpness]\label{thm:sharp}
If some $j\neq i$ has $|s_i-s_j|\le 2\varepsilon$, an admissible midpoint $\delta$ destroys uniqueness.
\end{theorem}

\section{Intuition}
An $\ell_\infty$ adversary moves any two scores by at most $2\varepsilon$ relative to each other. Gaps larger than $2\varepsilon$ cannot reverse or tie; gaps of size at most $2\varepsilon$ admit a midpoint collision.

\section{Examples}
\begin{example}
Scores $(1,4,7)$, $k=1$, $\varepsilon=1$: gaps $3,6,3$ all exceed $2$, so the middle index is stable. For $\varepsilon=2$, the gap $3\le 4$ admits a midpoint tie between the first two coordinates.
\end{example}

\section{Proof}
Let $s'=s+\delta$ with $\|\delta\|_\infty\le\varepsilon$.

\paragraph{Pairwise orders.}
If $s_p<s_q$ and $s_q-s_p>2\varepsilon$, then $s'_p-s'_q\le -(s_q-s_p)+2\varepsilon<0$, so $s'_p<s'_q$. The converse is symmetric. Hence $s_p<s_q\Leftrightarrow s'_p<s'_q$ whenever $|s_p-s_q|>2\varepsilon$; under all-gaps this holds for every distinct pair.

\paragraph{Count.}
Taking $q=i$ yields $\#\{j:s_j<s_i\}=\#\{j:s'_j<s'_i\}$. If the left side equals $k$, so does the right.

\paragraph{Uniqueness.}
If $s'_j=s'_i$ and $j\neq i$, all-gaps give $|s_j-s_i|>2\varepsilon$, hence $s'_j\neq s'_i$, contradiction. Thus $i$ remains the unique strict $k$-th index (Theorem~\ref{thm:inv}).

\paragraph{Sharpness.}
For $j\neq i$ with $|s_i-s_j|\le 2\varepsilon$, set $\delta_i=-(s_i-s_j)/2$, $\delta_j=(s_i-s_j)/2$, and $\delta_t=0$ otherwise. Then $\|\delta\|_\infty\le\varepsilon$ and $s'_i=s'_j$, so uniqueness fails (Theorem~\ref{thm:sharp}).
