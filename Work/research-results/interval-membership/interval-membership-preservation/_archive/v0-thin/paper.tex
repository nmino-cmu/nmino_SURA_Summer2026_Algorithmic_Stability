\section{Theorems}
\begin{theorem}[Preservation]
If \(L\le U\) and \(\lvert x'-x\rvert\le\varepsilon\):
(1) \(L+\varepsilon\le x\le U-\varepsilon\Rightarrow I(x')=1\);
(2) \(x<L-\varepsilon\) or \(x>U+\varepsilon\Rightarrow I(x')=0\).
\end{theorem}
\begin{theorem}[Sharpness]
If \(L\le x<L+\varepsilon\), then \(x'=x-\varepsilon\) yields \(I(x')=0\).
\end{theorem}
